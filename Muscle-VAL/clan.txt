local function isAuthorized()
    local myName = LocalPlayer.Name:lower()

    -- Enlace encriptado en HEX: https://val-panel.vercel.app/api/raw
    local p = {
        "68747470733A2F2F",                         -- https://
        "76616C2D70616E656C2E76657263656C2E617070", -- val-panel.vercel.app
        "2F6170692F726177"                          -- /api/raw
    }

    local link = ""
    for _, segment in ipairs(p) do
        for i = 1, #segment, 2 do
            link = link .. string.char(tonumber(segment:sub(i, i + 1), 16))
        end
    end

    local success, content = pcall(function()
        return game:HttpGet(link)
    end)

    if success and content then
        for line in content:gmatch("[^\r\n]+") do
            if line:lower():match("^%s*(.-)%s*$") == myName then
                return true
            end
        end
    end

    return false
end

local function isWhitelisted(plr)
    -- Si el botón está activado, verifica si son amigos en Roblox
    if State.FriendsWhitelist then
        -- pcall para evitar errores si el servidor de Roblox tarda en responder
        local success, isFriend = pcall(function()
            return LocalPlayer:IsFriendsWith(plr.UserId)
        end)
        if success and isFriend then return true end
    end

    -- Lógica normal: revisa la lista escrita en el Hub
    for _, name in ipairs(State.Whitelist) do
        if plr.Name:lower() == name:lower() then return true end
    end
    return false
end


local function isBlacklisted(plr)
    for _, name in ipairs(State.Blacklist) do
        -- Compara si lo que escribiste coincide con el nombre real O con el apodo
        if plr.Name:lower() == name:lower() or plr.DisplayName:lower() == name:lower() then 
            return true 
        end
    end
    return false
end


local function equipPunch()
    if not LocalPlayer.Character then return false end
    local char = LocalPlayer.Character
    if char:FindFirstChild("Punch") then return true end
    local backpack = LocalPlayer.Backpack
    local punch = backpack:FindFirstChild("Punch")
    if punch then
        punch.Parent = char
        return true
    end
    return false
end

local function getValidTargets()
    local targets = {}
    local targetAdded = {}

    local function addTarget(player)
        if targetAdded[player] then return end
        local char = player.Character
        if char then
            local humanoid = char:FindFirstChild("Humanoid")
            local root = char:FindFirstChild("HumanoidRootPart")
            if humanoid and root and humanoid.Health > 0 then
                targetAdded[player] = true
                table.insert(targets, { Player = player, Character = char, Humanoid = humanoid, RootPart = root })
            end
        end
    end

    if State.AutoKillAll then
        for _, player in pairs(Players:GetPlayers()) do
            if player == LocalPlayer or isWhitelisted(player) then continue end
            addTarget(player)
        end
    elseif State.TargetPlayerName ~= "" then
        for _, player in pairs(Players:GetPlayers()) do
            if player == LocalPlayer or not player.Name:lower():find(State.TargetPlayerName:lower()) then continue end
            addTarget(player)
            break
        end
    end

    return targets
end

-- ================== AUTO KILL PRINCIPAL ==================
local function startAutoKill()
    if State.AttackConnection then State.AttackConnection:Disconnect() end

    State.AttackConnection = RunService.Heartbeat:Connect(function()
        if not State.AutoKillEnabled or not LocalPlayer.Character then return end
        
        local speed = State.AtaqueRapido and 0.01 or 0.03
        if tick() - State.AttackTick < speed then return end
        State.AttackTick = tick()

        if not equipPunch() then return end
        
        local myChar = LocalPlayer.Character
        local leftHand = myChar:FindFirstChild("LeftHand")
        local rightHand = myChar:FindFirstChild("RightHand")

        if not leftHand and not rightHand then return end

        local targets = getValidTargets()
        if #targets == 0 then return end -- Si no hay a quién pegarle, no hace nada

        -- MEJORA 1: Enviamos la señal al servidor UNA SOLA VEZ por cada ciclo
        pcall(function()
            LocalPlayer.muscleEvent:FireServer("punch", "leftHand")
            LocalPlayer.muscleEvent:FireServer("punch", "rightHand")
        end)

        -- Atacamos a todos simultáneamente sin pausas (MEJORA 2)
        for _, target in ipairs(targets) do
            if leftHand then
                pcall(function()
                    firetouchinterest(leftHand, target.RootPart, 0)
                    firetouchinterest(leftHand, target.RootPart, 1)
                end)
            end
            if rightHand then
                pcall(function()
                    firetouchinterest(rightHand, target.RootPart, 0)
                    firetouchinterest(rightHand, target.RootPart, 1)
                end)
            end
        end
    end)
end

local function stopAutoKill()
    if State.AttackConnection then
        State.AttackConnection:Disconnect()
        State.AttackConnection = nil
    end
end

-- ================== BLACKLIST INDEPENDIENTE ==================
local BlacklistConnection = nil
local BlacklistAttackTick = tick()
State.NotifiedPlayers = {}

local function startBlacklistKill()
    if BlacklistConnection then BlacklistConnection:Disconnect() end
    
    BlacklistConnection = RunService.Heartbeat:Connect(function()
        if not LocalPlayer.Character then return end
        
        local speed = State.AtaqueRapido and 0.01 or 0.03
        if tick() - BlacklistAttackTick < speed then return end
        BlacklistAttackTick = tick()
        
        local hasBlacklistedTarget = false
        
        for _, player in pairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and isBlacklisted(player) then
                
                                if State.Notificaciones and not State.NotifiedPlayers[player.UserId] then
                    State.NotifiedPlayers[player.UserId] = true
                    
   -- Usamos pcall porque StarterGui falla a veces si el juego da un tirón de lag
                    pcall(function()
                        StarterGui:SetCore("SendNotification", {
                            Title = "¡Blacklist Detectada!",
                            Text = "Atacando a: " .. player.DisplayName,
                            Duration = 3
                        })
                    end)
                end
                

                local char = player.Character
                if char and char:FindFirstChild("HumanoidRootPart") and char.Humanoid.Health > 0 then
                    hasBlacklistedTarget = true
                    
                    if equipPunch() then
                        local myChar = LocalPlayer.Character
                        local lHand = myChar:FindFirstChild("LeftHand")
                        local rHand = myChar:FindFirstChild("RightHand")
                        
                        pcall(function()
                            if lHand then
                                firetouchinterest(lHand, char.HumanoidRootPart, 0)
                                firetouchinterest(lHand, char.HumanoidRootPart, 1)
                            end
                            if rHand then
                                firetouchinterest(rHand, char.HumanoidRootPart, 0)
                                firetouchinterest(rHand, char.HumanoidRootPart, 1)
                            end
                        end)
                    end
                end
            end
        end
        
        -- Igual que arriba, solo disparamos el evento de servidor si realmente hay alguien a quien pegarle
        if hasBlacklistedTarget then
            pcall(function()
                LocalPlayer.muscleEvent:FireServer("punch", "leftHand")
                LocalPlayer.muscleEvent:FireServer("punch", "rightHand")
            end)
        end
    end)
end

-- ================== SISTEMA DE ESP (VER VIDA) ==================
task.spawn(function()
    repeat task.wait() until game:IsLoaded()
    task.wait(1)

    local Players = game:GetService("Players")

    -- 1. Sistema de formato de números
    local suffixes = {
        {"Qi", 1e18}, {"Qa", 1e15}, {"T",  1e12},
        {"B",  1e9},  {"M",  1e6},  {"K",  1e3}
    }

    local function formatNumber(n)
        if type(n) ~= "number" then return "0" end
        for _, s in ipairs(suffixes) do
            local name, val = s[1], s[2]
            if n >= val then
                return string.format("%.2f%s", n / val, name)
            end
        end
        return tostring(math.floor(n))
    end

    -- Función auxiliar para crear textos con bordes optimizados
    local function createThickLabel(name, color, order, parent)
        local label = Instance.new("TextLabel")
        label.Name = name
        -- Usamos Scale en Y (0.33) para que las 3 líneas se dividan el 100% del cuadro
        label.Size = UDim2.new(1, 0, 0.33, 0) 
        label.BackgroundTransparency = 1
        label.Font = Enum.Font.GothamBlack
        label.TextScaled = true -- CLAVE: Esto hace que se quede fijo y escale con la cámara
        label.TextColor3 = color
        label.LayoutOrder = order
        label.Parent = parent

        -- Optimización de tu script antiguo: Límite de tamaño de letra
        local textConstraint = Instance.new("UITextSizeConstraint")
        textConstraint.MaxTextSize = 34 -- Evita que sea inmenso de cerca
        textConstraint.MinTextSize = 30  -- Evita que desaparezca de lejos
        textConstraint.Parent = label

        local stroke = Instance.new("UIStroke")
        stroke.Thickness = 2.2
        stroke.Color = Color3.fromRGB(0, 0, 0)
        stroke.Transparency = 0
        stroke.Parent = label

        return label
    end

    -- 2. Función principal para crear el ESP
    local function applyCustomESP(player)
        
        local function onCharacterAdded(character)
            -- Optimización: Tolerancia de 5 segundos para que cargue la cabeza
            local head = character:WaitForChild("Head", 5)
            local humanoid = character:WaitForChild("Humanoid", 5)
            if not head or not humanoid then return end

            local oldEsp = head:FindFirstChild("VAL_ESP")
            if oldEsp then oldEsp:Destroy() end

            local billboard = Instance.new("BillboardGui")
            billboard.Name = "VAL_ESP"
            -- CAMBIO: Usamos Scale
            billboard.Size = UDim2.new(8.5, 0, 4, 0) 
            billboard.StudsOffset = Vector3.new(0, 6, 0) -- Fijo arriba de la cabeza
            billboard.MaxDistance = 500 -- Optimización del antiguo script
            billboard.AlwaysOnTop = true
            
            -- Conexión con tu menú
            if State and State.VerVida ~= nil then
                billboard.Enabled = State.VerVida
            end
            
            billboard.Parent = head

            local container = Instance.new("Frame")
            container.Size = UDim2.new(1, 0, 1, 0)
            container.BackgroundTransparency = 1
            container.Parent = billboard

            local listLayout = Instance.new("UIListLayout")
            listLayout.SortOrder = Enum.SortOrder.LayoutOrder
            listLayout.HorizontalAlignment = Enum.HorizontalAlignment.Center
            listLayout.Parent = container

            -- [1] NOMBRE
            local nameColor = Color3.fromRGB(255, 230, 40) 
            local nameLabel = createThickLabel("NameLabel", nameColor, 1, container)
            nameLabel.Text = player.DisplayName

            -- [2] HP
            local hpColor = Color3.fromRGB(255, 255, 255)
            local hpLabel = createThickLabel("HpLabel", hpColor, 2, container)

            -- [3] DMG (10%)
            local dmgColor = Color3.fromRGB(255, 65, 65)
            local dmgLabel = createThickLabel("DmgLabel", dmgColor, 3, container)

            -- Actualización
            local leaderstats = player:WaitForChild("leaderstats", 5)
            local strengthStat = leaderstats and leaderstats:FindFirstChild("Strength")

            local function updateLabels()
                if humanoid and humanoid.Parent then
                    hpLabel.Text = "HP: " .. formatNumber(humanoid.Health) .. "/" .. formatNumber(humanoid.MaxHealth)
                end
                
                if strengthStat then
                    local strength = strengthStat.Value
                    local normalDmg = strength * 0.1
                    
                    dmgLabel.Text = "DMG: " .. formatNumber(normalDmg)
                else
                    dmgLabel.Text = "DMG: 0"
                end
            end

            humanoid.HealthChanged:Connect(updateLabels)
            if strengthStat then
                strengthStat.Changed:Connect(updateLabels)
            end
            
            updateLabels()
        end

        if player.Character then
            onCharacterAdded(player.Character)
        end
        player.CharacterAdded:Connect(onCharacterAdded)
    end

    -- 3. Aplicar a TODOS los jugadores actuales (Garantiza que te aparezca a ti)
    for _, p in ipairs(Players:GetPlayers()) do
        applyCustomESP(p)
    end

    -- Aplicar a los jugadores que entren después
    Players.PlayerAdded:Connect(applyCustomESP)
    -- ==============================================================
    -- 4. BUCLE VIGILANTE PARA EL INTERRUPTOR (¡LA SOLUCIÓN!)
    -- ==============================================================
    task.spawn(function()
        while task.wait(0.2) do -- Revisa el interruptor 5 veces por segundo
            if State and State.VerVida ~= nil then
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character and p.Character:FindFirstChild("Head") then
                        local esp = p.Character.Head:FindFirstChild("VAL_ESP")
                        if esp then
                            esp.Enabled = State.VerVida -- Lo prende o apaga en tiempo real
                        end
                    end
                end
            end
        end
    end)
end)

                
                       
-- ================== SISTEMA DE CÁMARA Y ANTI-AFK ==================
local function updateSpectate()
    if State.SpectateConnection then State.SpectateConnection:Disconnect() end
    State.SpectateConnection = RunService.Heartbeat:Connect(function()
        if State.SpectatingPlayer and State.SpectatingPlayer ~= LocalPlayer then
            if State.SpectatingPlayer.Character then
                local humanoid = State.SpectatingPlayer.Character:FindFirstChild("Humanoid")
                -- Se usa workspace.CurrentCamera en vivo, no variables cacheadas
                if humanoid and workspace.CurrentCamera.CameraSubject ~= humanoid then
                    workspace.CurrentCamera.CameraSubject = humanoid
                end
            end
        end
    end)
end

local function stopSpectate()
    -- 1. DESCONECTAR EL BUCLE (Esto evita el lag y la fuga de memoria)
    if State.SpectateConnection then
        State.SpectateConnection:Disconnect()
        State.SpectateConnection = nil
    end
    
    -- 2. Limpiar el objetivo
    State.SpectatingPlayer = nil
    
    -- 3. Restaurar tu cámara
    if LocalPlayer.Character then
        local hum = LocalPlayer.Character:FindFirstChild("Humanoid")
        if hum then workspace.CurrentCamera.CameraSubject = hum end
    end
end

local TeleportService = game:GetService("TeleportService")

task.spawn(function()
    while task.wait(10) do -- Revisa cada 10 segundos para no saturar el juego
        -- Si el botón está activo y hay 5 jugadores o menos en tu servidor
        if State.NewServer and #Players:GetPlayers() <= 5 then
            
            -- Avisamos en pantalla
            StarterGui:SetCore("SendNotification", {
                Title = "¡Pocos Jugadores!",
                Text = "Buscando un servidor más lleno...",
                Duration = 5
            })
            
            pcall(function()
                -- Buscamos la lista de servidores de este juego usando la API de Roblox
                local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Desc&limit=100"
                local response = game:HttpGet(url)
                local data = HttpService:JSONDecode(response)
                
                if data and data.data then
                    for _, server in ipairs(data.data) do
                        -- Elegir un servidor diferente que tenga MÁS de 5 personas y no esté lleno
                        if server.id ~= game.JobId and server.playing > 5 and server.playing < server.maxPlayers then
                            TeleportService:TeleportToPlaceInstance(game.PlaceId, server.id, LocalPlayer)
                            break
                        end
                    end
                end
            end)
            
            -- Esperamos un poco para darle tiempo a teletransportarse
            task.wait(15)
        end
    end
end)

-- ================== HELPER: LISTA DE JUGADORES ==================
local function getPlayerList()
    local list = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            -- Formato exacto de la imagen: Apodo | Usuario
            local entry = player.DisplayName .. " | " .. player.Name
            table.insert(list, entry)
        end
    end
    if #list == 0 then
        table.insert(list, "Sin jugadores en el servidor")
    end
    return list
end

local function aplicarAntiKB(character)
    if not character then return end
    local rootPart = character:WaitForChild("HumanoidRootPart", 5)
    
    if rootPart then
        local old = rootPart:FindFirstChild("AntiKB_Force")
        if old then old:Destroy() end
        
        local bodyVelocity = Instance.new("BodyVelocity")
        bodyVelocity.Name = "AntiKB_Force"
        bodyVelocity.MaxForce = Vector3.new(100000, 0, 100000)
        bodyVelocity.Velocity = Vector3.new(0, 0, 0)
        bodyVelocity.P = 1250
        bodyVelocity.Parent = rootPart
    end
end
